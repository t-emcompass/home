Split-Core RF current Probe · Radiated Emission Debugging

About this product

The T-EMCompass RF Current Probe is a compact split-core current probe designed for locating and characterising radiated emission sources during pre-compliance EMC debugging.

By clamping around a cable or PCB trace without disconnecting it, the probe measures the RF common-mode current — the primary driver of radiated emissions. This makes it an ideal first-step tool for identifying which cable or signal path is responsible for an emission, allowing engineers to prioritise filtering and layout improvements with confidence.

The probe is available as a calibrated, ready-to-use instrument or as an open-source DIY design for engineers who prefer to build their own. The housing is fully parametric in OpenSCAD and optimised for standard FDM 3D printing.

OpenSCAD design files

////////////////////////////////////////////////////////////
// T-EMCompass RF Current Probe — Split-Core Housing
// Split-core ferrite housing
//   - parametric core dimensions
//   - correct hinge alignment
//   - real cable feedthrough
//   - true full-moon split
//
// Part variable:
//   0 = assembly view
//   1 = half A (SMA side)
//   2 = half B
////////////////////////////////////////////////////////////

$fn = 120;

// ---- Core dimensions ----
core_od = 26.9;
core_id = 16.4;
core_h  = 30;

// ---- Clearances & walls ----
clear = 0.4;
wall  = 2.5;

cav_od = core_od + 2*clear;
cav_id = core_id - 2*clear;
height = core_h + (2*clear) + 5;
outer_od = cav_od + 2*wall;

// ---- Hinge ----
hinge_pin_d = 3.2;
hinge_lug_w = 6;
hinge_lug_h = 6;

// ---- Cable feedthrough ----
wire_d      = 1.0;
wire_clear  = 0.5;
wire_hole_d = wire_d + wire_clear;

// ---- SMA mount ----
sma_x = outer_od/2;
sma_y = 0;
sma_z = height/2;

// ---- Output selection ----
part = 0;  // 0=assembly, 1=halfA, 2=halfB


// ============================================================
// FULL RING (base geometry)
// ============================================================
module full_ring()
{
    difference()
    {
        cylinder(h=height, r=outer_od/2);

        translate([0, 0, 2.5])
            cylinder(h=height-5, r=cav_od/2);

        translate([0, 0, -1])
            cylinder(h=height+2, r=cav_id/2);

        translate([0, 0, 0])
            cylinder(h=height+2, r=wire_hole_d/2);
    }
}


// ============================================================
// HALF CUT (true moon split)
// ============================================================
module half(body=true)
{
    intersection()
    {
        full_ring();
        translate([-outer_od, -outer_od/2, -1])
            cube([outer_od, outer_od, height+2]);
    }
}


// ============================================================
// HINGE LUGS
// ============================================================
module rounded_lug_2d(w, h, r)
{
    hull()
    {
        translate([-w/2 + r, -h/3])  circle(r=r);
        translate([-w/2 + r,  h/2])  circle(r=r);
        translate([ w/2,     -h/2])  square([0.01, h]);
    }
}

module hinge_lugs(side=1)
{
    lug_w   = hinge_lug_w + 3;
    lug_h   = hinge_lug_h + 2;
    fillet_r = 2;

    translate([-1.7, side*((outer_od/2)+4.8), height/4])
    difference()
    {
        union()
        {
            linear_extrude(height=height/4, center=false)
                rounded_lug_2d(lug_w, lug_h, fillet_r);
            translate([0, 0, height/2])
                linear_extrude(height=height/4-2, center=false)
                    rounded_lug_2d(lug_w, lug_h, fillet_r);
        }
        translate([lug_w/3 - fillet_r*0.6, -1, 0])
            rotate([180,0,0])
                cylinder(h=lug_h*10, r=hinge_pin_d/2, center=true);
    }
}


// ============================================================
// SMA HOLE + SUPPORT BLOCK
// ============================================================
module sma_hole()
{
    translate([sma_x+15, sma_y, sma_z])
        rotate([0,90,0])
            cylinder(h=outer_od*2, r=6.5/2, center=true);
}

module sma_support_block()
{
    translate([outer_od*2-11, 0, height/2])
        cube([6, 13, 13], center=true);
}


// ============================================================
// HALF SHELL A
// ============================================================
module half_A()
{
    difference()
    {
        union() { half(); sma_support_block(); }
        sma_hole();
    }
    mirror([0,1,0]) hinge_lugs(-1);
}


// ============================================================
// HALF SHELL B
// ============================================================
module half_B()
{
    difference()
    {
        rotate([0,0,180]) half();
        sma_hole();
    }
    mirror([1,0,0]) hinge_lugs(-1);
}


// ============================================================
// ASSEMBLY VIEW
// ============================================================
module assembly()
{
    color("silver")    half_A();
    translate([outer_od+5, 0, 0])
        color("gainsboro") half_B();
}


// ============================================================
// OUTPUT
// ============================================================
if (part == 0) assembly();
if (part == 1) half_A();
if (part == 2) half_B();
